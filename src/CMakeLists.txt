qt_add_executable(quickshell main.cpp)

if (WIN32)
	set_target_properties(quickshell PROPERTIES OUTPUT_NAME qs)

	qt_add_executable(quickshell-gui main.cpp)
	set_target_properties(quickshell-gui PROPERTIES OUTPUT_NAME qsw WIN32_EXECUTABLE ON)
	target_compile_definitions(quickshell-gui PRIVATE QS_WINDOWS_GUI_EXE)

	set(QS_PRODUCT_NAME "Quickshell" CACHE STRING "Product name in the Windows version resource")
	set(QS_VERSION_FILE "" CACHE FILEPATH "File whose first line is the product version (default: the project version)")
	set(QS_PRODUCT_VERSION "${PROJECT_VERSION}")
	if (QS_VERSION_FILE AND EXISTS "${QS_VERSION_FILE}")
		file(STRINGS "${QS_VERSION_FILE}" QS_VERSION_LINES LIMIT_COUNT 1)
		string(STRIP "${QS_VERSION_LINES}" QS_PRODUCT_VERSION)
		set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${QS_VERSION_FILE}")
	endif()
	string(REGEX MATCH "^[0-9]+(\\.[0-9]+)*" QS_VERSION_NUMERIC "${QS_PRODUCT_VERSION}")
	string(REPLACE "." ";" QS_VERSION_PARTS "${QS_VERSION_NUMERIC}")
	list(LENGTH QS_VERSION_PARTS QS_VERSION_PART_COUNT)
	while (QS_VERSION_PART_COUNT LESS 4)
		list(APPEND QS_VERSION_PARTS 0)
		list(LENGTH QS_VERSION_PARTS QS_VERSION_PART_COUNT)
	endwhile()
	list(SUBLIST QS_VERSION_PARTS 0 4 QS_VERSION_PARTS)
	string(REPLACE ";" "," QS_RC_VERSION_NUMBERS "${QS_VERSION_PARTS}")
	configure_file(version_info.h.in "${CMAKE_CURRENT_BINARY_DIR}/version_info.h" @ONLY)

	target_sources(quickshell PRIVATE quickshell.rc)
	target_sources(quickshell-gui PRIVATE quickshell.rc)
	target_include_directories(quickshell PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
	target_include_directories(quickshell-gui PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")

	install(TARGETS quickshell quickshell-gui RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
else()
	install(TARGETS quickshell RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
